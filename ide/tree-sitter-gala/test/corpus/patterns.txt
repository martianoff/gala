package main

func f(x any) string = x match {
    case n: int if n > 0 => "positive"
    case 1 | 2 => "small"
    case Some("a" | "b") => "letter"
    case Cons(head, tail...) => "list"
    case Some(Right((a, b))) => "nested"
    case _ => {
        Println("other")
        "other"
    }
}
